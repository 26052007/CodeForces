import sys
 
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
 
    t = int(input_data[0])
    results = []
 
    for i in range(1, t + 1):
        n = int(input_data[i])
        ans = 0
        l = 1
        while l <= n:
            v = n // l
            r = n // v
            ans += v * v * (r - l + 1)
            l = r + 1
        results.append(str(ans))
 
    sys.stdout.write("
".join(results) + "
")
 
 
if __name__ == "__main__":
    solve()
