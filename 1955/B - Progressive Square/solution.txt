import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    t = int(input_data[0])
    idx = 1
    
    results = []
    for _ in range(t):
        n = int(input_data[idx])
        c = int(input_data[idx+1])
        d = int(input_data[idx+2])
        idx += 3
        
        b = [int(x) for x in input_data[idx:idx + n*n]]
        idx += n * n
        
        a11 = min(b)
        expected = []
        for i in range(n):
            for j in range(n):
                expected.append(a11 + i * c + j * d)
                
        b.sort()
        expected.sort()
        
        if b == expected:
            results.append("YES")
        else:
            results.append("NO")
            
    sys.stdout.write('
'.join(results) + '
')
 
if __name__ == '__main__':
    solve()
