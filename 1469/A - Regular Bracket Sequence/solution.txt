import sys
 
def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    t = int(data[0])
    results = []
    
    for i in range(1, t + 1):
        s = data[i]
        if len(s) % 2 != 0 or s[0] == ')' or s[-1] == '(':
            results.append("NO")
        else:
            results.append("YES")
            
    print('
'.join(results))
 
if __name__ == '__main__':
    solve()
