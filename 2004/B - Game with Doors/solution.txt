import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    t = int(input_data[0])
    idx = 1
    results = []
    
    for _ in range(t):
        l = int(input_data[idx])
        r = int(input_data[idx+1])
        L = int(input_data[idx+2])
        R = int(input_data[idx+3])
        idx += 4
        
        start = max(l, L)
        end = min(r, R)
        
        if start > end:
            results.append("1")
        else:
            ans = end - start
            if min(l, L) < start:
                ans += 1
            if max(r, R) > end:
                ans += 1
            results.append(str(ans))
            
    sys.stdout.write('
'.join(results) + '
')
 
if __name__ == '__main__':
    solve()
