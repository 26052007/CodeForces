import sys
 
def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    l = int(data[1])
    a = sorted([int(x) for x in data[2:n+2]])
    
    max_gap = 0
    for i in range(n - 1):
        gap = a[i+1] - a[i]
        if gap > max_gap:
            max_gap = gap
            
    ans = max(a[0], l - a[-1], max_gap / 2.0)
    print(f"{ans:.10f}")
 
if __name__ == '__main__':
    solve()
