import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    t = int(input_data[0])
    idx = 1
    results = []
    
    for _ in range(t):
        n = int(input_data[idx])
        m = int(input_data[idx+1])
        k = int(input_data[idx+2])
        idx += 3
        
        sa = {int(x) for x in input_data[idx:idx+n] if int(x) <= k}
        idx += n
        
        sb = {int(x) for x in input_data[idx:idx+m] if int(x) <= k}
        idx += m
        
        if len(sa) >= k // 2 and len(sb) >= k // 2 and len(sa | sb) == k:
            results.append("YES")
        else:
            results.append("NO")
            
    sys.stdout.write('
'.join(results) + '
')
 
if __name__ == '__main__':
    solve()
