import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    t = int(input_data[0])
    idx = 1
    out = []
    
    for _ in range(t):
        n = int(input_data[idx])
        idx += 1
        
        ans = 0
        last = -float('inf')
        
        for _ in range(n):
            val = int(input_data[idx])
            idx += 1
            if val >= last + 2:
                ans += 1
                last = val
                
        out.append(str(ans))
        
    sys.stdout.write('
'.join(out) + '
')
 
if __name__ == '__main__':
    solve()
